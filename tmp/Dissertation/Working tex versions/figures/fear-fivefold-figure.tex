\begin{tikzpicture}[

  agent/.style={draw=orange!75!black, line width=1pt, fill=orange!12,

% --- centre: the agent / orexis (efficient cause) ---
\node[agent] (agent) at (0,0)
  {\textbf{the agent}\\[1pt]\emph{fear befalls}\\[1pt]
   \textit{orexis}\,---\,moved mover\\
   {\footnotesize\itshape\textcolor{orange!75!black}{efficient cause}}};

% --- four corner facets ---
\node[facet] (disp) at (-4.3,2.75)
  {\textbf{\cnum{1} Disposition}\\
   finding-oneself-\emph{threatened}; the \emph{involuntary} uptake---one must
   ``believe'' the threat is \emph{for me, now}\\[1pt]
   {\footnotesize\itshape\textcolor{blue!60}{formal cause: \textmd{\textit{doxa}}'s evaluative form}}};

\node[facet] (intent) at (4.3,2.75)
  {\textbf{\cnum{2} Intentionality}\\
   \emph{toward-which}: the destructive evil (the figure, the weapon)\\
   \emph{toward-whom}: the assailant\,\textperiodcentered\,\textit{Mitsein}\\[1pt]
   {\footnotesize\itshape\textcolor{blue!60}{final cause: the good sought (safety)}}};

\node[facet] (body) at (4.3,-2.75)
  {\textbf{\cnum{3} Bodily form}\\
   the \emph{cold} of fear---\gk{ταραχή}, chilled blood, flight-readiness\\[1pt]
   {\footnotesize\itshape\textcolor{blue!60}{material cause: the affections (\textmd{\textit{DMA}})}}};

\node[facet] (disc) at (-4.3,-2.75)
  {\textbf{\cnum{5} Disclosive function}\\
   brings-to-\textit{krisis}\,/\,\textit{logos}: discloses the world as
   threatening; opens flight, defence, or seeking-help};

% --- faint links: facets of ONE gestalt, not stages ---
\draw[link] (agent) -- (disp);
\draw[link] (agent) -- (intent);
\draw[link] (agent) -- (body);

% --- the befalling, from without ---
\node[font=\itshape, text=red!65, align=center] (fromw) at (0,4.75)

\node[align=center, font=\small] (hb) at (-4.1,\hy-0.66)
  {\textbf{having-been} (\textit{Gewesenheit})\\ sedimented \gk{οἴεσθαι}\\ the thrown ground};

\caption{\textbf{Fear as a cohesive unfolding that befalls the agent.} The five-fold
structure (\cnum{1}--\cnum{5}) and the four causes are set out as distinct facets only
to exhibit their linkage; the \textit{pathos} is \emph{lived} as a single co-disclosed
event, not assembled stage-by-stage. The agent-centre and facets \cnum{1}--\cnum{3}
carry the four causes---\emph{efficient} (\textit{orexis}, the moved mover),
\emph{formal} (\textit{doxa}'s evaluative form), \emph{final} (the apparent good,
safety), and \emph{material} (the affections)---the structure inherited from the
\textit{Metaphysics}, \textit{Rhetoric}, and \textit{De Anima}; \cnum{4} temporal range
and \cnum{5} disclosive function are the two moments the analysis develops \emph{beyond}
the inherited schemata. Fear is \emph{primarily futural} (\gk{μέλλον}), grounded in the
having-been of the dispositional \gk{οἴεσθαι} and felt in the present \gk{ταραχή}---the
ecstatic unity of the three (\textit{BT}~\S68b). The dotted radii and the inward
(\textit{paschein}) arrows mark the involuntary character: fear arrives \emph{from
without} and \emph{from the futural}, not as the agent's stage-by-stage construction.}
\label{fig:fear-fivefold}
